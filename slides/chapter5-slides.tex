\begin{frame}{5.1 Introduction}
    \textbf{Guardrails in Medical AI:}
    \begin{itemize}
        \item Critical protective layer for medical assistants
        \item Prevents harmful, incorrect, or unethical content
        \item Ensures compliance with medical standards
    \end{itemize}
    
    \vspace{1em}
    
    \textbf{Key Objectives:}
    \begin{itemize}
        \item Input filtering
        \item Output monitoring
        \item Specialized medical policies enforcement
    \end{itemize}
\end{frame}

\begin{frame}{5.2 Medical Domain Challenges}
    \textbf{Specific Challenges:}
    \begin{itemize}
        \item \textbf{Unauthorized diagnosis:} System must not replace doctors
        \item \textbf{Drug prescriptions:} No specific treatment recommendations
        \item \textbf{PHI protection:} HIPAA compliance requirements
        \item \textbf{Emergency situations:} Proper escalation procedures
        \item \textbf{Drug interactions:} Preventing dangerous advice
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Implementation Goals:}
    \begin{itemize}
        \item User protection
        \item System credibility
        \item Legal compliance
        \item Risk management
        \item Trust building
    \end{itemize}
\end{frame}

    \textbf{Multi-layer Architecture:}

    \end{itemize}
\end{frame}

\begin{frame}{5.4 Input Security Policies}
    \textbf{Nine Key Input Controls:}
    
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{enumerate}
            \item No personal diagnosis requests
            \item Protection of safety rules
            \item Emergency situation management
            \item Sensitive content filtering
            \item Abusive language prevention
        \end{enumerate}
        
        \column{0.5\textwidth}
        \begin{enumerate}
            \setcounter{enumi}{5}
            \item PHI protection
            \item Identity impersonation prevention
            \item System security
            \item Malicious code blocking
        \end{enumerate}
    \end{columns}
    
    \vspace{0.5em}
    
    \begin{center}
        \colorbox{red!20}{\textbf{Blocks inappropriate requests before processing}}
    \end{center}
\end{frame}

\begin{frame}{5.5 Output Security Policies}
    \textbf{Eight Core Output Criteria:}
    
    \begin{itemize}
        \item \textbf{Harm prevention:} No dangerous medical advice
        \item \textbf{Professional disclaimer:} Not a doctor replacement
        \item \textbf{Treatment limitations:} No specific recommendations
        \item \textbf{Information accuracy:} No medical misinformation
        \item \textbf{Self-diagnosis prevention:} No encouragement of self-treatment
        \item \textbf{Evidence-based:} Avoid unverified claims
        \item \textbf{Appropriate tone:} Clear, empathetic, cautious
        \item \textbf{Proper referrals:} Direct to professionals when needed
    \end{itemize}
    
    \begin{center}
        \colorbox{green!20}{\textbf{Ensures safe and appropriate responses}}
    \end{center}
\end{frame}

\begin{frame}{5.6 Technical Challenges}
    \textbf{Implementation Challenges:}
    \begin{itemize}
        \item Balance between safety and usability
        \item \textcolor{red}{176\% increase in response time} (additional 730ms)
        \item Comprehensive scenario coverage
        \item Bilingual support (Persian/English)
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Current Limitations:}
    \begin{itemize}
        \item Context detection challenges
        \item Increased computational load
        \item Need for continuous policy updates
        \item Cultural adaptation requirements
    \end{itemize}
\end{frame}
