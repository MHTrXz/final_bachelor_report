\begin{frame}{Medical LLMs \& Persian Research}
    \textbf{A. Advances in Medical LLMs}
    \begin{itemize}
        \item Transformer-based models: GPT, PaLM, Gemma, Med-PaLM
        \item Specialized models: Med-PaLM 2, MEDITRON, BioMistral, Aloe Family, Meerkat
        \item Retrieval-Augmented Generation (RAG) to enhance factuality
        \item Fine-tuning: Supervised, LoRA/QLoRA/DLoRA for efficiency
        \item Advanced reasoning: Chain-of-Thought, Tree of Thoughts, self-consistency
    \end{itemize}
    \vspace{0.5em}
    \textbf{B. Persian Medical AI}
    \begin{itemize}
        \item Limited research; English-centric models dominate
        \item Introduction of MF3QA and Persian medical corpora
        \item Growing need for native Persian solutions
    \end{itemize}
\end{frame}

\begin{frame}{Techniques, Tools, and Innovations}
    \textbf{A. Core Techniques}
    \begin{itemize}
        \item Supervised fine-tuning with domain data
        \item RAG with late chunking for improved context
        \item Safety and guardrails: NeMo, two-layer architecture
        \item Preference tuning: RLHF, DPO, KTO, ORPO
    \end{itemize}
    \vspace{0.5em}
    \textbf{B. Tools \& Infrastructures}
    \begin{itemize}
        \item ElasticSearch, Guardrails, LangChain, Ollama
        \item Unsloth, TensorBoard, Lightning.ai, Vast.ai
    \end{itemize}
    \vspace{0.5em}
    \textbf{C. Project Innovations}
    \begin{enumerate}
        \item Late chunking in RAG for full context
        \item Two-layer guardrails for medical safety
        \item Multilingual (Persian-English) optimization
    \end{enumerate}
\end{frame}
