\begin{frame}{Late Chunking Technique}
    \textbf{Traditional vs Late Chunking:}
    \begin{itemize}
        \item \textbf{Traditional:} Chunk → Embed → Store
        \item \textbf{Late Chunking:} Full Document → Embed → Chunk → Store
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Three-Phase Process:}
    \begin{enumerate}
        \item \textbf{Recursive Chunking:} Split by natural separators with size control
        \item \textbf{Simultaneous Vector Generation:} Process all chunks together
        \item \textbf{Aggregation and Pooling:} Generate unified representation
    \end{enumerate}
    
    \vspace{0.5em}
    
    \begin{center}
        \colorbox{highlight!20}{\textbf{Key Benefit: Full context preserved in each chunk}}\\
        \vspace{0.3em}
        \colorbox{improvement!20}{\textbf{Result: 15-20\% improvement in retrieval accuracy}}
    \end{center}
\end{frame}

\begin{frame}{Persian Processing and Cross-lingual Challenges}
    \textbf{Persian Processing with Hazm:}
    \begin{itemize}
        \item Text normalization and sentence boundary detection
        \item Max chunk: 500 chars, Overlap: 100 chars
        \item Generated 5,176,679 chunks from 102K documents
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Cross-lingual Retrieval Challenge:}
    \begin{itemize}
        \item Users ask in Persian, documents in English (and vice versa)
        \item Traditional methods (TF-IDF, BM25) fail
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Solution:}
    \begin{center}
        \colorbox{highlight!20}{\textbf{Cosine similarity on multilingual embeddings}}
    \end{center}
    
    Benefits: Cross-lingual semantic understanding, unified knowledge base
\end{frame}
