\begin{frame}{3.1 Introduction}
    \textbf{Supervised Fine-Tuning:}
    \begin{itemize}
        \item Key technique for optimizing LLMs for specialized applications
        \item Enables adaptation of pre-trained models to specific domains
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Chapter Objectives:}
    \begin{itemize}
        \item Adapt pre-trained models to medical domain
        \item Fine-tune four models: Aya-expanse-8b, Gemma-2-9B, Gemma-3-12B-IT, gpt-oss-20b
        \item Train on both English and Persian medical data
        \item Optimize LoRA parameters for efficient training
        \item Evaluate and compare performance comprehensively
    \end{itemize}

\begin{frame}{3.4 Initial Testing with Default Configuration}

\begin{frame}{3.5 LoRA Parameter Optimization}
    \textbf{Optimization Methodology:}
    \begin{itemize}
        \item Fast testing on Aya-Expanse-8B with 50K samples subset
        \item Testing rank/alpha combinations: 16, 32, 64, 128
        \item Evaluation on MedQA accuracy
    \end{itemize}
    
    \vspace{0.5em}
    
    \begin{table}[h]
        \centering
        \small
        \caption{Table 3.7: LoRA Parameter Test Results}
        \begin{tabular}{l|ccc}
            \toprule
            \textbf{Rank-Alpha} & \textbf{(k,q,v,o)} & \textbf{(k,q,v,o,up,down,gate)} & \textbf{(all)} \\
            \midrule
            16-16 & 51.0\% & 53.7\% & \textcolor{improvement}{55.41\%} \\
            32-32 & 50.6\% & 50.36\% & 52.7\% \\
            64-64 & 50.2\% & 50.0\% & 50.5\% \\
            128-128 & 48.21\% & 43.27\% & 42.40\% \\
            \bottomrule
        \end{tabular}
    \end{table}
    
    \textbf{Key Finding:} rank=32, alpha=32 selected for full training
\end{frame}

\begin{frame}{3.9 Persian Training Motivation}
    \textbf{Necessity:}
    \begin{itemize}
        \item Lack of specialized Persian medical models
        \item Limited Persian medical resources
        \item Need for culturally appropriate medical AI
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Approach:}
    \begin{itemize}
        \item Continue from English-trained models (transfer learning)
        \item Leverage English medical knowledge for Persian adaptation
        \item Address Persian-specific medical terminology and context
    \end{itemize}
\end{frame}

\begin{frame}{3.10 Persian Dataset: MF3QA}
    \textbf{First comprehensive Persian medical QA dataset:}
    
    \vspace{0.5em}
    
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Dataset Composition:}
        \begin{itemize}
            \item 20,000 training samples
            \item 2,000 development samples
            \item 2,000 test samples
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Training Strategy:}
        \begin{itemize}
            \item Merge train + dev = 22K samples
            \item Use test set for evaluation
            \item Focus on quality over quantity
        \end{itemize}
    \end{columns}
    
    \vspace{1em}
    
    \textbf{Evaluation Datasets:}
    \begin{itemize}
        \item Persian translations of MMLU medical subsets
        \item Same 5 categories as English evaluation
    \end{itemize}
\end{frame}

\begin{frame}{3.11 Persian Model Selection}
    \textbf{Selection Criteria:}
    \begin{itemize}
        \item Persian language support capability
        \item Acceptable performance in initial evaluations
        \item Good English training results (if previously trained)
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Selected Models:}
    \begin{enumerate}
        \item \textbf{gpt-oss-20b}: High capacity model with good Persian support
        \item \textbf{Gemma-3-12b-it}: Best performer in English training
    \end{enumerate}
\end{frame}

\begin{frame}{3.12 Persian Training Configuration}
    \textbf{Key Differences from English Training:}
    \begin{itemize}
        \item Much smaller dataset (22K vs 200K)
        \item Lower LoRA parameters to prevent overfitting
        \item rank=4, alpha=8 (vs rank=32, alpha=32 for English)
    \end{itemize}

            \item Epochs: 3
            \item Batch size: 32
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{GPT-OSS-20B Configuration:}
        \begin{itemize}
            \item Dataset: MF3QA (22K)
            \item LoRA rank: 4
            \item LoRA alpha: 8
            \item Epochs: 3

\end{frame}

\begin{frame}{3.15 Chapter Summary}

    \begin{itemize}
