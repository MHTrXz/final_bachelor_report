\begin{frame}{Dataset Quality Optimization}
    \begin{columns}[T]
        \column{0.6\textwidth}
        \textbf{Filtering Process:}
        \begin{itemize}
            \item Remove incorrect/ambiguous answers
            \item Remove duplicate questions
            \item Remove unreliable sources
            \item Remove off-topic content
        \end{itemize}
        
        \column{0.4\textwidth}
        \textbf{Results:}
        \begin{itemize}
            \item Original: 410K samples
            \item Filtered: 200K samples
            \item Quality: 3.2/5 → 4.1/5
            \item \textcolor{improvement}{28\% quality improvement}
        \end{itemize}
    \end{columns}
    
    \vspace{0.5em}
    
    \textbf{Final Training Configuration:}
    \begin{itemize}
        \item Model: Gemma-3-12B-IT
        \item LoRA: rank=32, alpha=32, target=all modules
        \item Optimizer: adamw-8bit
        \item Quantization: 4bit
        \item Epochs: 3, Learning rate: 2e-4
    \end{itemize}
\end{frame}

\begin{frame}{Persian Training Motivation}
    \textbf{Challenges:}
    \begin{itemize}
        \item Lack of specialized Persian medical models
        \item Limited Persian medical resources
        \item Need for culturally appropriate medical AI
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Dataset: MF3QA}
    \begin{itemize}
        \item First comprehensive Persian medical QA dataset
        \item 22,000 training samples (train+dev)
        \item 2,000 test samples for evaluation
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Approach:}
    \begin{itemize}
        \item Transfer learning from English models
        \item Conservative LoRA parameters (rank=4, alpha=8)
        \item Prevent overfitting on limited data
    \end{itemize}
\end{frame}

\begin{frame}{Persian Model Selection and Configuration}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Selected Models:}
        \begin{itemize}
            \item \textbf{gpt-oss-20b}: High capacity with Persian support
            \item \textbf{Gemma-3-12b-it}: Best English performer
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Configuration:}
        \begin{itemize}
            \item Dataset: MF3QA (22K)
            \item LoRA rank: 4
            \item LoRA alpha: 8
            \item Epochs: 3
            \item Learning rate: 2e-4
        \end{itemize}
    \end{columns}
    
    \vspace{0.5em}
    
    \textbf{Key Differences from English:}
    \begin{itemize}
        \item Much smaller dataset (22K vs 200K)
        \item Lower LoRA parameters (4/8 vs 32/32)
        \item Focus on preventing overfitting
    \end{itemize}
\end{frame}
