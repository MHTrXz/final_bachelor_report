\begin{frame}{6.2 Key Achievements - English}
    \textbf{Fine-tuning Results:}
    \begin{itemize}
        \item Model: Gemma-3-12B-IT with 200K medical samples
        \item Best improvement: MedQA with +9.22\% accuracy
        \item Final model: gemma-3-12b-it-en-sft
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{RAG System:}
    \begin{itemize}
        \item 373.5K specialized documents (StatePearls + Textbooks)
        \item Late Chunking technique implementation
        \item Additional improvement: up to 2.29\% in some domains
    \end{itemize}
\end{frame}

\begin{frame}{6.3 Key Achievements - Persian}
    \textbf{Model Development:}
    \begin{itemize}
        \item Two models trained: gpt-oss-20b and Gemma-3-12B-IT
        \item Dataset: MF3QA (22K samples)
        \item Best model: gemma-3-12b-it-fa-sft (70.8\% accuracy)
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Bilingual RAG:}
    \begin{itemize}
        \item Persian Medical Corpus: 5,176,679 chunks
        \item Cross-lingual search capability
        \item Comprehensive bilingual platform
    \end{itemize}
\end{frame}

\begin{frame}{6.6 Future Work - Priority 1}
    \textbf{Reinforcement Learning Implementation:}
    
    \begin{itemize}
        \item Improve response quality through user feedback
        \item Adapt to user preferences
        \item Reduce inappropriate responses
        \item Ensure balanced bilingual quality
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Implementation Plan:}
    \begin{enumerate}
        \item Collect bilingual expert feedback
        \item Design reward function (accuracy, safety, usefulness)
        \item Apply PPO or DPO techniques
        \item Separate evaluation for each language
        \item Continuous improvement system
    \end{enumerate}
\end{frame}

\begin{frame}{6.7 Future Work - Priority 2}
    \textbf{Guardrails Optimization:}
    
    Target: Reduce response time to <500ms while maintaining >90\% filtering accuracy
    
    \begin{itemize}
        \item \textbf{Smaller models:} DistilBERT (English), ParsBERT (Persian)
        \item \textbf{Parallel processing:} Async pipeline optimization
        \item \textbf{Smart caching:} Semantic hashing for similar requests
        \item \textbf{Complexity reduction:} Rule-based for simple cases
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{RAG Enhancement:}
    \begin{itemize}
        \item Fine-tune embedder for medical domain
        \item Implement Graph RAG
        \item Expand Persian knowledge base
        \item Balance bilingual resources
    \end{itemize}
\end{frame}
