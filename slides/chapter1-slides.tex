\begin{frame}{Recent Advances in LLMs}
    \textbf{Significant Progress in Large Language Models:}
    \begin{itemize}
        \item Deep architectures and extensive text data enable natural language understanding
        \item Models like GPT, LLaMA, and T5 perform diverse tasks with just prompts
        \item Remarkable capabilities in general language tasks
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Challenges in Specialized Applications:}
    \begin{itemize}
        \item Limited access to organizational and up-to-date specialized data
        \item Risk of inaccurate or dangerous responses in sensitive domains
        \item Particularly critical in healthcare and legal applications
    \end{itemize}
\end{frame}

\begin{frame}{1.1 Project Objectives}
    \textbf{Goal:} Design and implement a medical language understanding assistant based on LLMs
    
    \vspace{0.5em}
    
    \textbf{Key Features:}
    \begin{itemize}
        \item \textbf{Accurate Responses:} Specialized content-appropriate answers
        \item \textbf{Knowledge Base Access:} Internal organizational data integration
        \item \textbf{Safety Mechanisms:} Control and filtering for quality assurance
        \item \textbf{Domain Adaptability:} Healthcare, education, customer support
    \end{itemize}
    
    \vspace{0.5em}
    
    \textbf{Three Main Components:}
    \begin{enumerate}
        \item Fine-tuning with specialized medical data
        \item RAG architecture for internal knowledge base
        \item Control mechanisms for safety and accuracy
    \end{enumerate}
\end{frame}

\begin{frame}{1.2 Dual-Track Project Structure}
    \textbf{Two Parallel Development Paths:}
    
    \vspace{1em}
    
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Track 1: English Model}
        \begin{itemize}
            \item English model with Persian support
            \item Fine-tuning on English medical data
            \item RAG implementation for bilingual environments
            \item Primary development path
        \end{itemize}
        
        \column{0.5\textwidth}
        \textbf{Track 2: Persian Model}
        \begin{itemize}
            \item Dedicated Persian model development
            \item Persian-specific fine-tuning
            \item Persian information retrieval
            \item Comparative evaluation
        \end{itemize}
    \end{columns}
    
    \vspace{1em}
    
    \begin{center}
        \colorbox{highlight!20}{Enables comparison and analysis of both approaches}
    \end{center}
\end{frame}
