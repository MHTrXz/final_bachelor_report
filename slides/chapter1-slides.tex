\begin{frame}{Introduction \& Objectives}
    \textbf{Persian Medical Assistant Based on Large Language Models}
    \begin{itemize}
        \item \textbf{Goal:} Design and implement an intelligent assistant for Persian medical language understanding and specialized response generation.
        \item \textbf{Features:}
        \begin{itemize}
            \item Accurate, context-aware answers
            \item Access to up-to-date and organizational data
            \item Safety mechanisms and quality control
            \item Adaptability to healthcare, education, and more
        \end{itemize}
        \item \textbf{Innovation:} Dual-track development—English model with Persian support and a dedicated Persian model, enabling comparative analysis.
    \end{itemize}
\end{frame}

\begin{frame}{Related Work}
    \textbf{Key Research Areas:}
    \begin{itemize}
        \item Large Language Models (LLMs): GPT, LLaMA, T5
        \item Domain adaptation and supervised fine-tuning for medical applications
        \item Retrieval-Augmented Generation (RAG) methods
        \item AI for healthcare and medical information systems
        \item Multilingual and Persian-specific language modeling
        \item Safety, guardrails, and response filtering in sensitive domains
        \item Integration of organizational data with LLMs
    \end{itemize}
\end{frame}

\begin{frame}{Project Structure}
    \textbf{Dual Development Tracks:}
    \begin{itemize}
        \item English-based model (with Persian support)
        \item Dedicated Persian model
    \end{itemize}
    \vspace{1em}
    \textbf{Main Components:}
    \begin{enumerate}
        \item Fine-tuning with specialized medical data
        \item RAG (Retrieval-Augmented Generation) architecture for internal knowledge base
        \item Safety and control mechanisms (Guardrails)
    \end{enumerate}
    \vspace{1em}
    \textbf{Outcome:} The first practical Persian LLM medical assistant with advanced safety features and real-world applicability.
\end{frame}
